'use client';

import React, {
  useCallback,
  useEffect,
  useRef,
  useState,
  useMemo,
} from 'react';
import type { OnMount, BeforeMount } from '@monaco-editor/react';
import type * as Monaco from 'monaco-editor';
import dynamic from 'next/dynamic';
import { CodeEditorProps, Language } from './types';
import { EditorToolbar } from './EditorToolbar';
import { registerCustomTheme } from './lib/monacoTheme';
import { registerPythonProvider } from './lib/pythonProvider';
import { formatCode } from './lib/formatters';
import styles from './CodeEditor.module.css';

// Lazy-load Monaco Editor — avoids SSR and reduces initial bundle
const MonacoEditor = dynamic(
  async () => {
    const { default: Editor } = await import('@monaco-editor/react');
    return { default: Editor };
  },
  {
    ssr: false,
    loading: () => (
      <div className={styles.editorSkeleton} aria-busy aria-label="Loading editor">
        <div className={styles.skeletonLine} style={{ width: '60%' }} />
        <div className={styles.skeletonLine} style={{ width: '80%' }} />
        <div className={styles.skeletonLine} style={{ width: '45%' }} />
        <div className={styles.skeletonLine} style={{ width: '70%' }} />
        <div className={styles.skeletonLine} style={{ width: '55%' }} />

    ),
  },
);

let pythonProviderRegistered = false;

export const CodeEditor: React.FC<CodeEditorProps> = ({
  value,
  defaultValue,
  language: initialLanguage = 'javascript',
  onChange,
  readOnly = false,
  height = '520px',
  className,
}:{
value?: string;
defaultValue?: Record<Language, string>;
language?: Language;
onChange?: (value: string) => void;
readOnly?: boolean;
height?: number | string;
className?: string;
}) => {
  const [language, setLanguage] = useState<Language>(initialLanguage);
  const [isFormatting, setIsFormatting] = useState(false);
  const [isCopied, setIsCopied] = useState(false);
  const [isEditorReady, setIsEditorReady] = useState(false);
  const editorRef = useRef<Monaco.editor.IStandaloneCodeEditor | null>(null);
  const monacoRef = useRef<typeof Monaco | null>(null);
  const disposablesRef = useRef<Monaco.IDisposable[]>([]);
  const copyTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  // Stable ref so handleEditorMount (empty-dep callback) always calls the
  // latest handleFormat without needing it in its dependency array.
  const handleFormatRef = useRef<() => Promise<void>>(async () => {});

  const isControlled = value !== undefined;
  const [internalValue, setInternalValue] = useState<string>(
    () => defaultValue?.[initialLanguage] ?? '',
  );

  const currentValue = isControlled ? value : internalValue;

  // Sync readOnly option changes without re-mounting
  useEffect(() => {
    editorRef.current?.updateOptions({ readOnly });
  }, [readOnly]);

  // Sync language changes
  useEffect(() => {
    if (!editorRef.current || !monacoRef.current) return;
    const model = editorRef.current.getModel();
    if (model) {
      monacoRef.current.editor.setModelLanguage(
        model,
        language === 'javascript' ? 'javascript' : 'python',
      );
    }
  }, [language]);

  const handleBeforeMount: BeforeMount = useCallback((monaco) => {
    registerCustomTheme(monaco);

    if (!pythonProviderRegistered) {
      registerPythonProvider(monaco);
      pythonProviderRegistered = true;
    }

    // JS language service: syntax highlighting + completions only.
    // Semantic validation (TypeScript compiler worker) is explicitly OFF —
    // this editor never compiles or type-checks user code.
    monaco.languages.typescript.javascriptDefaults.setDiagnosticsOptions({
      noSemanticValidation: true,   // OFF — no TS compiler runs
      noSyntaxValidation: false,    // ON  — tokeniser-level parse errors only
      noSuggestionDiagnostics: true, // OFF — no compiler-driven suggestions
    });

    // Minimal options so the worker starts without invoking the compiler.
    // noEmit + noCheck means zero compilation output and zero type-checking.
    monaco.languages.typescript.javascriptDefaults.setCompilerOptions({
      target: monaco.languages.typescript.ScriptTarget.ESNext,
      allowNonTsExtensions: true,
      allowJs: true,
      checkJs: false,   // explicitly no type-checking
      noEmit: true,     // no output, ever
      noResolve: true,  // no module resolution (prevents phantom FS access)
    });
  }, []);

  const handleEditorMount: OnMount = useCallback(
    (editor, monaco) => {
      editorRef.current = editor;
      monacoRef.current = monaco;
      setIsEditorReady(true);

      // Keyboard shortcut: Alt+Shift+F → Format.
      // editor.addCommand() returns string|null (the command ID), NOT an
      // IDisposable — commands are owned by the editor and freed with it.
      // Call via ref so this empty-dep callback always invokes the latest
      // handleFormat without it ever appearing in the dependency array.
      editor.addCommand(
        monaco.KeyMod.Alt | monaco.KeyMod.Shift | monaco.KeyCode.KeyF,
        () => void handleFormatRef.current(),
      );

      // Selection glow via CSS class injection
      editor.onDidChangeCursorSelection(() => {
        const sel = editor.getSelection();
        const hasSelection =
          sel !== null &&
          (sel.startLineNumber !== sel.endLineNumber ||
            sel.startColumn !== sel.endColumn);
        editor.getDomNode()?.classList.toggle('has-selection', hasSelection);
      });

      // Auto-resize whenever the container dimensions change
      const resizeObserver = new ResizeObserver(() => editor.layout());
      const container = editor.getDomNode()?.parentElement;
      if (container) resizeObserver.observe(container);
      disposablesRef.current.push({ dispose: () => resizeObserver.disconnect() });
    },
    // Deliberately empty: all mutable values are accessed through refs.
    [],
  );

  const handleFormat = useCallback(async () => {
    const editor = editorRef.current;
    if (!editor) return;
    setIsFormatting(true);
    try {
      const raw = editor.getValue();
      const formatted = await formatCode(raw, language);

      // Preserve cursor position
      const position = editor.getPosition();
      editor.setValue(formatted);
      if (position) editor.setPosition(position);
      editor.revealPositionInCenter(editor.getPosition()!);

      if (!isControlled) setInternalValue(formatted);
      onChange?.(formatted);
    } catch (err) {
      console.error('[CodeEditor] Format error:', err);
    } finally {
      setIsFormatting(false);
    }
  },[isControlled, language, onChange]);

  useEffect(() => {
    handleFormatRef.current = handleFormat;
  }, [handleFormat]);

  const handleCopy = useCallback(async () => {
    const text = editorRef.current?.getValue() ?? currentValue ?? '';
    try {
      await navigator.clipboard.writeText(text);
    } catch {
      // Fallback for browsers that block clipboard without user gesture
      const ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed';
      ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.focus();
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
    }
    setIsCopied(true);
    if (copyTimerRef.current) clearTimeout(copyTimerRef.current);
    copyTimerRef.current = setTimeout(() => setIsCopied(false), 2000);
  }, [currentValue]);

  const handleLanguageChange = useCallback(
    (lang: Language) => {
      setLanguage(lang);
      if (!isControlled) {
        if (defaultValue) {
          setInternalValue(defaultValue[lang] ?? '');
        } else {
          setInternalValue('');
        }
      }
    },
    [defaultValue, isControlled],
  );

  const handleChange = useCallback(
    (val: string | undefined) => {
      const newVal = val ?? '';
      if (!isControlled) setInternalValue(newVal);
      onChange?.(newVal);
    },
    [isControlled, onChange],
  );

  // Cleanup
  useEffect(() => {
    const disposables = disposablesRef.current;
    return () => {
      disposables.forEach((d) => d.dispose());
      if (copyTimerRef.current) clearTimeout(copyTimerRef.current);
    };
  }, []);

  const editorOptions = useMemo<Monaco.editor.IStandaloneEditorConstructionOptions>(
    () => ({
      readOnly,
      fontSize: 13.5,
      fontFamily: '"JetBrains Mono", "Fira Code", "Cascadia Code", "SF Mono", monospace',
      fontLigatures: true,
      lineHeight: 22,
      letterSpacing: 0.3,
      minimap: { enabled: false },
      scrollBeyondLastLine: false,
      padding: { top: 20, bottom: 20 },
      lineNumbers: 'on',
      lineNumbersMinChars: 3,
      glyphMargin: true,
      folding: true,
      foldingHighlight: true,
      lineDecorationsWidth: 6,
      renderLineHighlight: 'gutter',
      cursorBlinking: 'phase',
      cursorSmoothCaretAnimation: 'on',
      cursorStyle: 'line',
      cursorWidth: 2,
      smoothScrolling: true,
      contextmenu: true,
      suggestOnTriggerCharacters: true,
      quickSuggestions: { other: true, comments: false, strings: true },
      quickSuggestionsDelay: 100,
      wordBasedSuggestions: 'allDocuments',
      parameterHints: { enabled: true, cycle: true },
      formatOnPaste: false,
      formatOnType: false,
      autoClosingBrackets: 'languageDefined',
      autoClosingQuotes: 'languageDefined',
      autoSurround: 'languageDefined',
      autoIndent: 'full',
      tabSize: language === 'python' ? 4 : 2,
      insertSpaces: true,
      detectIndentation: false,
      wordWrap: 'on',
      wordWrapColumn: 100,
      wrappingIndent: 'indent',
      bracketPairColorization: { enabled: true },
      guides: {
        bracketPairs: 'active',
        bracketPairsHorizontal: 'active',
        indentation: true,
        highlightActiveBracketPair: true,
      },
      inlayHints: { enabled: 'on', fontSize: 11 },
      hover: { enabled: true, delay: 300 },
      occurrencesHighlight: 'singleFile',
      selectionHighlight: true,
      renderWhitespace: 'selection',
      renderControlCharacters: false,
      scrollbar: {
        vertical: 'auto',
        horizontal: 'auto',
        useShadows: false,
        verticalScrollbarSize: 5,
        horizontalScrollbarSize: 5,
        arrowSize: 0,
      },
      overviewRulerBorder: false,
      overviewRulerLanes: 2,
      hideCursorInOverviewRuler: false,
      suggest: {
        showKeywords: true,
        showSnippets: true,
        showFunctions: true,
        showVariables: true,
        showModules: true,
        showClasses: true,
        showProperties: true,
        preview: true,
        previewMode: 'subwordSmart',
        insertMode: 'insert',
        filterGraceful: true,
        shareSuggestSelections: false,
      },
      acceptSuggestionOnCommitCharacter: true,
      acceptSuggestionOnEnter: 'smart',
    }),
    [readOnly, language],
  );

  return (
    <div className={`${styles.container} font-mono ${className ?? ''}`}>
      {/* Glassmorphism ambient glow orbs */}
      <div className={styles.glowOrb1} aria-hidden /> 
      <div className={styles.glowOrb2} aria-hidden />

      <div className={styles.editorFrame}>
        {/* Traffic-light dots */}
        <div className={styles.windowChrome} aria-hidden>
          <span className={`${styles.chromeDot} ${styles.chromeDotRed}`} />
          <span className={`${styles.chromeDot} ${styles.chromeDotYellow}`} />
          <span className={`${styles.chromeDot} ${styles.chromeDotGreen}`} />
          <span className={styles.chromeTitle}>
            {language === 'javascript' ? 'index.js' : 'main.py'}
          </span>
        </div>

        <EditorToolbar
          language={language}
          onLanguageChange={handleLanguageChange}
          onFormat={handleFormat}
          onCopy={handleCopy}
          isFormatting={isFormatting}
          isCopied={isCopied}
          readOnly={readOnly}
          isReady={isEditorReady}
        />

        <div
          className={styles.editorWrapper}
          style={{ height: typeof height === 'number' ? `${height}px` : height }}
        >
          <MonacoEditor
            height="100%"
            language={language === 'javascript' ? 'javascript' : 'python'}
            value={currentValue}
            theme="neon-dark"
            beforeMount={handleBeforeMount}
            onMount={handleEditorMount}
            onChange={handleChange}
            options={editorOptions}
            loading={null}
          />
        </div>

        {/* Bottom status bar */}
        <div className={styles.statusBar} role="status" aria-live="polite">

// filepath: src/components/CodeEditor/lib/formatters.ts

// IMPORTANT: This module is a FORMATTER ONLY — it does NOT compile, execute,
// or evaluate any code. Prettier standalone runs entirely in-browser via AST
// parse → re-print; the Python formatter is pure string manipulation.
// No subprocess, no eval(), no Function(), no Worker with user code.
//
import type { Language } from '../types';

// ─── JavaScript Formatter (Prettier Standalone, browser-only) ────────────────
// Uses `prettier/standalone` which is a self-contained browser build with no
// Node.js APIs, no fs access, and no code execution — it only parses to an
// AST and reprints it.

async function formatJavaScript(code: string): Promise<string> {
  // Explicit plugin list prevents Prettier from loading any auto-detected
  // plugins that could involve native binaries or network access.
  const [prettier, babelPlugin, estreePlugin] = await Promise.all([
    import('prettier/standalone'),
    import('prettier/plugins/babel'),
    import('prettier/plugins/estree'),
  ]);

  return prettier.format(code, {
    parser: 'babel',                 // AST-only parse, no execution
    plugins: [
      babelPlugin.default  ?? babelPlugin,
      estreePlugin.default ?? estreePlugin,
    ],
    semi: true,
    singleQuote: true,
    jsxSingleQuote: false,
    trailingComma: 'all',
    printWidth: 80,
    tabWidth: 2,
    useTabs: false,
    bracketSpacing: true,
    bracketSameLine: false,
    arrowParens: 'always',
    endOfLine: 'lf',
  });
}

// ─── Python Formatter (in-browser, PEP-8 subset) ─────────────────────────────
// Strategy: preserve the author's indentation structure — Python's indentation
// is syntactically meaningful and cannot be safely recomputed without a real
// parser. We only do things that are safe on raw text:
//   1. Tabs → 4 spaces (PEP-8 §indentation)
//   2. Strip trailing whitespace on every line
//   3. Collapse >2 consecutive blank lines to 2 (PEP-8 §blank lines)
//   4. Ensure exactly 2 blank lines before top-level def/class/async def
//   5. Ensure exactly 1 blank line before nested def/class (indented)
//   6. Spacing around binary operators outside of string literals
//   7. Single trailing newline

function detectIndentUnit(lines: string[]): string {
  for (const line of lines) {
    const m = line.match(/^( +|\t+)/);
    if (!m) continue;
    if (m[0].includes('\t')) return '    ';
    // Take the leading-space count of the first indented line as the unit
    return ' '.repeat(m[0].length);
  }
  return '    ';
}

function normalizeTabs(line: string, indentUnit: string): string {
  // Replace only leading tabs; preserve tabs inside strings (best-effort)
  return line.replace(/^\t+/, (tabs) => indentUnit.repeat(tabs.length));
}

function getIndentDepth(line: string): number {
  const m = line.match(/^( *)/);
  return m ? m[1].length : 0;
}

function spaceOperators(line: string): string {
  // Only touch the non-string parts of the line (split on string literals)
  // We handle single-quoted, double-quoted, but NOT triple-quoted here
  // (triple-quoted lines are skipped by the caller).
  const parts = line.split(/("""[^"]*"""|'''[^']*'''|"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')/);
  return parts
    .map((part, i) => {
      // Even-indexed parts are outside strings
      if (i % 2 !== 0) return part;
      return part
        // Ensure space around ==, !=, <=, >=  (not already spaced)
        .replace(/([^\s=!<>])(==|!=|<=|>=)([^\s=])/g, '$1 $2 $3')
        // Ensure space around = assignment (not inside default args or type hints)
        // Only when not preceded by [=!<>+-*/] and not followed by =
        .replace(/([^\s=!<>+\-*/%&|^~,:(])=(?!=)/g, '$1 =')
        .replace(/=(?!=)([^\s=>])/g, '= $1')
        // Space after comma (if missing)
        .replace(/,(?=[^\s])/g, ', ')
        // Collapse multiple spaces (outside string literals)
        .replace(/([^\s]) {2,}([^\s])/g, '$1 $2');
    })
    .join('');
}

function formatPython(code: string): string {
  const rawLines = code.split('\n');
  const indentUnit = detectIndentUnit(rawLines);

  // Pass 1: normalize tabs + strip trailing whitespace
  const lines = rawLines.map((l) => normalizeTabs(l, indentUnit).trimEnd());

  // Track multi-line string state across lines
  let inMultilineString = false;
  let mlDelim = '';

  // Pass 2: operator spacing (skip multiline string contents)
  const spaced = lines.map((line) => {
    const stripped = line.trim();

    if (inMultilineString) {
      if (line.includes(mlDelim)) inMultilineString = false;
      return line; // preserve verbatim
    }

    // Detect opening of a multiline string on this line
    const dq = (line.match(/"""/g) ?? []).length;
    const sq = (line.match(/'''/g) ?? []).length;
    if (dq % 2 !== 0) { inMultilineString = true; mlDelim = '"""'; }
    else if (sq % 2 !== 0) { inMultilineString = true; mlDelim = "'''"; }

    if (stripped === '') return '';
    return spaceOperators(line);
  });

  // Pass 3: blank-line rules
  // Reset multiline tracker for second pass
  inMultilineString = false;
  mlDelim = '';

  const result: string[] = [];

  for (let i = 0; i < spaced.length; i++) {
    const line = spaced[i];
    const stripped = line.trim();

    // Track multiline strings
    if (inMultilineString) {
      if (line.includes(mlDelim)) inMultilineString = false;
      result.push(line);
      continue;
    }
    const dq = (line.match(/"""/g) ?? []).length;
    const sq = (line.match(/'''/g) ?? []).length;
    if (dq % 2 !== 0) { inMultilineString = true; mlDelim = '"""'; }
    else if (sq % 2 !== 0) { inMultilineString = true; mlDelim = "'''"; }

    // Blank line: pass through (we'll deduplicate below)
    if (stripped === '') {
      result.push('');
      continue;
    }

    // Is this line a def/class/async def?
    const isTopLevelDef = /^(def |class |async def )/.test(stripped);
    const isNestedDef   = /^ +(def |class |async def )/.test(line);
    const depth         = getIndentDepth(line);

    if (isTopLevelDef && result.length > 0) {
      // Ensure exactly 2 blank lines before top-level def/class
      while (result.length > 0 && result[result.length - 1] === '') result.pop();
      result.push('', '');
    } else if (isNestedDef && depth > 0 && result.length > 0) {
      // Ensure exactly 1 blank line before nested def/class
      // (only if the previous non-blank line is not an opening decorator)
      const prevNonBlank = [...result].reverse().find((l) => l.trim() !== '');
      if (prevNonBlank && !prevNonBlank.trim().startsWith('@')) {
        while (result.length > 0 && result[result.length - 1] === '') result.pop();
        result.push('');
      }
    }

    result.push(line);
  }

  // Pass 4: collapse >2 consecutive blank lines
  const collapsed: string[] = [];
  let blankRun = 0;
  for (const line of result) {
    if (line === '') {
      blankRun++;
      if (blankRun <= 2) collapsed.push('');
    } else {
      blankRun = 0;
      collapsed.push(line);
    }
  }

  // Trim leading/trailing blank lines, append single newline
  let out = collapsed.join('\n').trim();
  if (out) out += '\n';
  return out;
}

export async function formatCode(code: string, language: Language): Promise<string> {
  switch (language) {
    case 'javascript':
      return formatJavaScript(code);
    case 'python':
      return formatPython(code);
    default:
      return code;
  }

function makePythonDiagnostics(
  monaco: typeof Monaco,
  model: Monaco.editor.ITextModel,
): Monaco.editor.IMarkerData[] {
  const markers: Monaco.editor.IMarkerData[] = [];
  const lines = model.getLinesContent();

  lines.forEach((line, idx) => {
    const lineNumber = idx + 1;
    const trimmed = line.trim();

    // Detect common mistakes
    if (/^print\s+[^(]/.test(trimmed)) {
      markers.push({
        severity: monaco.MarkerSeverity.Error,
        message: 'SyntaxError: Missing parentheses in call to \'print\'',
        startLineNumber: lineNumber,
        startColumn: 1,
        endLineNumber: lineNumber,
        endColumn: line.length + 1,
        source: 'Python LSP',
      });
    }

    if (/^\s*(def|class)\s+\w.*[^:]$/.test(trimmed)) {
      markers.push({
        severity: monaco.MarkerSeverity.Error,
        message: 'SyntaxError: Expected \':\' at end of compound statement',
        startLineNumber: lineNumber,
        startColumn: 1,
        endLineNumber: lineNumber,
        endColumn: line.length + 1,
        source: 'Python LSP',
      });
    }

    if (/===|!==/.test(line)) {
      const col = line.indexOf('===') !== -1 ? line.indexOf('===') : line.indexOf('!==');
      markers.push({
        severity: monaco.MarkerSeverity.Warning,
        message: 'Python uses == and != for equality, not === or !==',
        startLineNumber: lineNumber,
        startColumn: col + 1,
        endLineNumber: lineNumber,
        endColumn: col + 4,
        source: 'Python LSP',
      });
    }

    if (/\bvar\s+\w+/.test(trimmed)) {
      markers.push({
        severity: monaco.MarkerSeverity.Warning,
        message: 'Python does not use \'var\' for variable declarations',
        startLineNumber: lineNumber,
        startColumn: 1,
        endLineNumber: lineNumber,
        endColumn: line.length + 1,
        source: 'Python LSP',
      });
    }

    // Trailing whitespace hint
    if (/\s+$/.test(line)) {
      markers.push({
        severity: monaco.MarkerSeverity.Hint,
        message: 'Trailing whitespace',
        startLineNumber: lineNumber,
        startColumn: line.trimEnd().length + 1,
        endLineNumber: lineNumber,
        endColumn: line.length + 1,
        source: 'Python LSP',
      });
    }
  });

  return markers;
}

    });
  });
}
